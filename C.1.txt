BLINK OF LED

#include "main.h"

int main(void)
{
    HAL_Init();

    SystemClock_Config();

    MX_GPIO_Init();

    while (1)
    {
        GPIOA->ODR |= (1U << 5);   // LED ON: set bit 5 using OR
        HAL_Delay(1000);           // LED ON for 1 second

        GPIOA->ODR &= ~(1U << 5);  // LED OFF: clear bit 5 using AND and NOT
        HAL_Delay(1000);           // LED OFF for 1 second
    }
}
